---
title: ImageSampler
---

Sampling parameters applied when drawing an image: wrapping on each axis
and filtering. [ImageSampler](/scripting/animascript/api-reference/image/image-sampler) is a value type, built with
`ImageSampler(wrapX, wrapY, filter)`. See also [ImageWrap](/scripting/animascript/api-reference/image/image-wrap) and
[ImageFilter](/scripting/animascript/api-reference/image/image-filter).
```as
export class DrawImage extends Node {
    myImage: Image? = null;
    sampler: ImageSampler = ImageSampler(ImageWrap.clamp, ImageWrap.clamp, ImageFilter.bilinear);

    override init(context: Context): bool {
        this.myImage = context.image("myImage");
        return true;
    }

    override draw(renderer: Renderer): void {
        let image = this.myImage;
        if (image != null) {
            renderer.drawImage(image, this.sampler, BlendMode.srcOver, 1);
        }
    }
}
```
